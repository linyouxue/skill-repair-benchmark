    public static class StageOrFinishEvent {
        public static final int KIND_SUBMIT = 0;
        public static final int KIND_FINISH = 1;

        public long jobId;
        public long timestampMs;
        public int kind;

        public StageOrFinishEvent() {}

        public StageOrFinishEvent(long jobId, long timestampMs, int kind) {
            this.jobId = jobId;
            this.timestampMs = timestampMs;
            this.kind = kind;
        }

        public static StageOrFinishEvent submit(long jobId, long timestampMs) {
            return new StageOrFinishEvent(jobId, timestampMs, KIND_SUBMIT);
        }

        public static StageOrFinishEvent finish(long jobId, long timestampMs) {
            return new StageOrFinishEvent(jobId, timestampMs, KIND_FINISH);
        }
    }

    public static class LongestStageProcess extends KeyedProcessFunction<Long, StageOrFinishEvent, Tuple2<Long, Integer>> {
        private final long gapMs;

        private transient ValueState<Long> lastSubmitTs;
        private transient ValueState<Integer> currentStageCount;
        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Boolean> finished;

        public LongestStageProcess(long gapMs) {
            this.gapMs = gapMs;
        }

        @Override
        public void open(Configuration parameters) {
            lastSubmitTs = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTs", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            finished = getRuntimeContext().getState(new ValueStateDescriptor<>("finished", Boolean.class));
        }

        @Override
        public void processElement(StageOrFinishEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            if (Boolean.TRUE.equals(finished.value())) {
                return;
            }

            if (event.kind == StageOrFinishEvent.KIND_SUBMIT) {
                handleSubmit(event);
                return;
            }

            flushCurrentStage();

            Integer max = maxStageCount.value();
            if (max == null) {
                max = 0;
            }
            out.collect(Tuple2.of(ctx.getCurrentKey(), max));

            lastSubmitTs.clear();
            currentStageCount.clear();
            maxStageCount.clear();
            finished.update(true);
        }

        private void handleSubmit(StageOrFinishEvent event) throws Exception {
            Integer count = currentStageCount.value();
            if (count == null) {
                count = 0;
            }

            Long last = lastSubmitTs.value();
            if (last == null) {
                currentStageCount.update(1);
                lastSubmitTs.update(event.timestampMs);
                return;
            }

            if (event.timestampMs > last && event.timestampMs - last > gapMs) {
                flushCurrentStage();
                currentStageCount.update(1);
            } else {
                currentStageCount.update(count + 1);
            }
            lastSubmitTs.update(Math.max(last, event.timestampMs));
        }

        private void flushCurrentStage() throws Exception {
            Integer current = currentStageCount.value();
            if (current == null || current == 0) {
                return;
            }

            Integer max = maxStageCount.value();
            if (max == null || current > max) {
                maxStageCount.update(current);
            }
        }
    }

    public static class MergedGzipEventsSource extends RichSourceFunction<StageOrFinishEvent> {
        private final String taskPath;
        private final String jobPath;
        private transient volatile boolean running = true;

        public MergedGzipEventsSource(String taskPath, String jobPath) {
            this.taskPath = taskPath;
            this.jobPath = jobPath;
        }

        @Override
        public void run(SourceContext<StageOrFinishEvent> ctx) throws Exception {
            try (BufferedReader taskReader = openGzip(taskPath);
                 BufferedReader jobReader = openGzip(jobPath)) {

                Cursor nextTask = readNextTaskSubmit(taskReader);
                Cursor nextJob = readNextJobFinish(jobReader);

                while (running && (nextTask != null || nextJob != null)) {
                    boolean takeTask;
                    if (nextJob == null) {
                        takeTask = true;
                    } else if (nextTask == null) {
                        takeTask = false;
                    } else {
                        takeTask = nextTask.timestampUs <= nextJob.timestampUs;
                    }

                    StageOrFinishEvent outEvent;
                    if (takeTask) {
                        outEvent = StageOrFinishEvent.submit(nextTask.jobId, nextTask.timestampUs / 1000L);
                        nextTask = readNextTaskSubmit(taskReader);
                    } else {
                        outEvent = StageOrFinishEvent.finish(nextJob.jobId, nextJob.timestampUs / 1000L);
                        nextJob = readNextJobFinish(jobReader);
                    }

                    synchronized (ctx.getCheckpointLock()) {
                        ctx.collect(outEvent);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }

        private static BufferedReader openGzip(String path) throws Exception {
            InputStream in = Files.newInputStream(Paths.get(path));
            GZIPInputStream gzipIn = new GZIPInputStream(in);
            return new BufferedReader(new InputStreamReader(gzipIn, StandardCharsets.UTF_8));
        }

        private Cursor readNextTaskSubmit(BufferedReader reader) throws Exception {
            String line;
            while (running && (line = reader.readLine()) != null) {
                if (line.isEmpty()) {
                    continue;
                }
                String[] p = line.split(",", -1);
                int eventType = Integer.parseInt(p[5]);
                if (eventType != TaskEvent.EVENT_SUBMIT) {
                    continue;
                }

                long tsUs = Long.parseLong(p[0]);
                long jobId = Long.parseLong(p[2]);
                return new Cursor(tsUs, jobId);
            }
            return null;
        }

        private Cursor readNextJobFinish(BufferedReader reader) throws Exception {
            String line;
            while (running && (line = reader.readLine()) != null) {
                if (line.isEmpty()) {
                    continue;
                }
                String[] p = line.split(",", -1);
                int eventType = Integer.parseInt(p[3]);
                if (eventType != JobEvent.EVENT_FINISH) {
                    continue;
                }

                long tsUs = Long.parseLong(p[0]);
                long jobId = Long.parseLong(p[2]);
                return new Cursor(tsUs, jobId);
            }
            return null;
        }

        private static class Cursor {
            final long timestampUs;
            final long jobId;

            Cursor(long timestampUs, long jobId) {
                this.timestampUs = timestampUs;
                this.jobId = jobId;
            }
        }
    }

    public static class LocalTextFileSink extends RichSinkFunction<String> {
        private final String outputPath;
        private transient BufferedWriter writer;

        public LocalTextFileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            File out = new File(outputPath);
            File parent = out.getParentFile();
            if (parent != null) {
                parent.mkdirs();
            }

            writer = new BufferedWriter(new OutputStreamWriter(new FileOutputStream(out, false), StandardCharsets.UTF_8));
        }

        @Override
        public void invoke(String value, Context context) throws Exception {
            writer.write(value);
            writer.newLine();
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.flush();
                writer.close();
            }
        }
