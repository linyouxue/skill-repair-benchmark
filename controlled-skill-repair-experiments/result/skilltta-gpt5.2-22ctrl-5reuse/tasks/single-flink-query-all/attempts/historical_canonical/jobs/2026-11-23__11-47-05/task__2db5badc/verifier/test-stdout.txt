                env.addSource(jobSourceOrTest(new JobEventFileSource(jobInput))).name("job-events");

                .<UnifiedEvent>forMonotonousTimestamps()
                .withTimestampAssigner((e, previousTimestamp) -> e.timestampMs);

                .filter(e -> e.eventType == TASK_EVENT_TYPE_SUBMIT)
                .map((MapFunction<TaskEvent, UnifiedEvent>) e -> UnifiedEvent.taskSubmit(e.jobId, e.timestampMs()))
                .assignTimestampsAndWatermarks(wm)
                .name("task-submit-events");

        DataStream<UnifiedEvent> finishEvents = jobEventsStream
                .filter(e -> e.eventType == JOB_EVENT_TYPE_FINISH)
                .map((MapFunction<JobEvent, UnifiedEvent>) e -> UnifiedEvent.jobFinish(e.jobId, e.timestampMs()))
                .assignTimestampsAndWatermarks(wm)
                .name("job-finish-events");

        DataStream<Tuple2<Long, Integer>> longestStagePerFinishedJob = submitEvents
                .union(finishEvents)
                .keyBy(e -> e.jobId)

    private enum EventKind {
        TASK_SUBMIT,
        JOB_FINISH
    }

    private static final class UnifiedEvent {
        public final long jobId;
        public final long timestampMs;
        public final EventKind kind;

        private UnifiedEvent(long jobId, long timestampMs, EventKind kind) {
            this.jobId = jobId;
            this.timestampMs = timestampMs;
            this.kind = kind;
        }

        public static UnifiedEvent taskSubmit(long jobId, long timestampMs) {
            return new UnifiedEvent(jobId, timestampMs, EventKind.TASK_SUBMIT);
        }

        public static UnifiedEvent jobFinish(long jobId, long timestampMs) {
            return new UnifiedEvent(jobId, timestampMs, EventKind.JOB_FINISH);
        }
    }

    private static final class LongestStagePerJobProcessFunction
            extends KeyedProcessFunction<Long, UnifiedEvent, Tuple2<Long, Integer>> {

        private final long gapMs;

        private transient ValueState<Long> lastSubmitTs;
        private transient ValueState<Integer> currentStageCount;
        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Long> finishTs;
        private transient ValueState<Boolean> emitted;

        private LongestStagePerJobProcessFunction(long gapMs) {
            this.gapMs = gapMs;
        }

        @Override
        public void open(Configuration parameters) {
            lastSubmitTs = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTs", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            finishTs = getRuntimeContext().getState(new ValueStateDescriptor<>("finishTs", Long.class));
            emitted = getRuntimeContext().getState(new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement(UnifiedEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Boolean alreadyEmitted = emitted.value();
            if (alreadyEmitted != null && alreadyEmitted) {
                return;
            }

            if (event.kind == EventKind.TASK_SUBMIT) {
                handleTaskSubmit(event, ctx);
            } else if (event.kind == EventKind.JOB_FINISH) {
                handleJobFinish(event, ctx);
            }
        }

        private void handleTaskSubmit(UnifiedEvent event, Context ctx) throws Exception {
            Long last = lastSubmitTs.value();
            Integer cur = currentStageCount.value();
            Integer max = maxStageCount.value();

            if (cur == null) {
                cur = 0;
            }
            if (max == null) {
                max = 0;
            }

            if (last != null && (event.timestampMs - last) > gapMs) {
                // Close previous stage.
                if (cur > max) {
                    max = cur;
                }
                cur = 0;
            }

            cur += 1;
            lastSubmitTs.update(event.timestampMs);
            currentStageCount.update(cur);
            maxStageCount.update(max);
        }

        private void handleJobFinish(UnifiedEvent event, Context ctx) throws Exception {
            Long existingFinish = finishTs.value();
            if (existingFinish == null || event.timestampMs < existingFinish) {
                if (existingFinish != null) {
                    ctx.timerService().deleteEventTimeTimer(existingFinish);
                }
                finishTs.update(event.timestampMs);
                ctx.timerService().registerEventTimeTimer(event.timestampMs);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Long f = finishTs.value();
            if (f == null || f != timestamp) {
                return;
            }

            Integer cur = currentStageCount.value();
            Integer max = maxStageCount.value();
            if (cur == null) {
                cur = 0;
            }
            if (max == null) {
                max = 0;
            }

            if (cur > max) {
                max = cur;
            }

            out.collect(Tuple2.of(ctx.getCurrentKey(), max));

            emitted.update(true);
            // Clear other states to free memory.
            lastSubmitTs.clear();
            currentStageCount.clear();
            maxStageCount.clear();
            finishTs.clear();
        }
    }

    private static final class TaskEventFileSource implements SourceFunction<TaskEvent> {
        private final String path;
        private volatile boolean running = true;

        private TaskEventFileSource(String path) {

            try (BufferedReader br = new BufferedReader(new InputStreamReader(openPossiblyGz(path), StandardCharsets.UTF_8))) {
                String line;
                while (running && (line = br.readLine()) != null) {
                    if (line.isEmpty()) {
                        continue;
                    }
                    ctx.collect(TaskEvent.fromCsvLine(line));
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    private static InputStream openPossiblyGz(String path) throws IOException {

    private static final class TextFileSink extends RichSinkFunction<String> {
        private final String outputPath;
        private transient BufferedWriter writer;

        private TextFileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            super.open(parameters);
            Path p = Path.of(outputPath);
            Path parent = p.getParent();
            if (parent != null) {
                Files.createDirectories(parent);
            }
            writer = Files.newBufferedWriter(
                    p,
                    StandardCharsets.UTF_8,
                    StandardOpenOption.CREATE,
                    StandardOpenOption.TRUNCATE_EXISTING,
                    StandardOpenOption.WRITE);
        }

        @Override
        public synchronized void invoke(String value, Context context) throws Exception {
            writer.write(value);
            writer.newLine();
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.flush();
                writer.close();
            }
            super.close();
        }
    }
