    private static WatermarkStrategy<TaskEvent> taskWatermarks() {

    private static long toMillis(long timestampMicros) {
        if (timestampMicros == Long.MAX_VALUE) {
            return Long.MAX_VALUE;

        return Math.floorDiv(timestampMicros, MICROS_PER_MILLISECOND);

    private static long timerMillis(long timestampMicros) {
        if (timestampMicros >= Long.MAX_VALUE - (MICROS_PER_MILLISECOND - 1)) {

    private static final class LongestStageProcessFunction
            extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, String> {
        private final long sessionMicros;
        private transient ListState<Long> submitTimestamps;
        private transient ValueState<Long> finishTimestamp;
        private transient ValueState<Boolean> emitted;

        private LongestStageProcessFunction(long sessionMicros) {
            this.sessionMicros = sessionMicros;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submit timestamps", Long.class));
            finishTimestamp = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("finish timestamp", Long.class));
            emitted = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("result emitted", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent event, Context context, Collector<String> out)
                throws Exception {
            if (event.getEventType() != SUBMIT || Boolean.TRUE.equals(emitted.value())) {
                return;
            }
            Long finishedAt = finishTimestamp.value();
            if (finishedAt == null || event.getTimestamp() <= finishedAt) {
                submitTimestamps.add(event.getTimestamp());
            }
        }

        @Override
        public void processElement2(JobEvent event, Context context, Collector<String> out)
                throws Exception {
            if (event.getEventType() != FINISH || Boolean.TRUE.equals(emitted.value())
                    || event.getTimestamp() == Long.MAX_VALUE) {
                return;
            }

            Long previousFinish = finishTimestamp.value();
            if (previousFinish == null || event.getTimestamp() < previousFinish) {
                finishTimestamp.update(event.getTimestamp());
                context.timerService().registerEventTimeTimer(timerMillis(event.getTimestamp()));
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext context, Collector<String> out)
                throws Exception {
            if (Boolean.TRUE.equals(emitted.value())) {
                return;
            }

            Long finishedAt = finishTimestamp.value();
            if (finishedAt == null) {
                return;
            }

            List<Long> timestamps = new ArrayList<>();
            for (Long submitTimestamp : submitTimestamps.get()) {
                if (submitTimestamp != null && submitTimestamp <= finishedAt) {
                    timestamps.add(submitTimestamp);
                }
            }
            if (!timestamps.isEmpty()) {
                Collections.sort(timestamps);
                long longestStage = longestStage(timestamps);
                out.collect("(" + context.getCurrentKey() + "," + longestStage + ")");
            }
            emitted.update(true);
        }

        private long longestStage(List<Long> sortedTimestamps) {
            long longest = 0;
            long current = 0;
            Long previous = null;
            for (long timestamp : sortedTimestamps) {
                if (previous == null || timestamp - previous >= sessionMicros) {
                    current = 1;
                } else {
                    current++;
                }
                longest = Math.max(longest, current);
                previous = timestamp;
            }
            return longest;
        }
