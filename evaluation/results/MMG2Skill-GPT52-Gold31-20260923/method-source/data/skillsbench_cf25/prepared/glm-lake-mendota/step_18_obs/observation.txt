recs=[]
for t_idx,h in enumerate(time_hours):
    dt=start+pd.Timedelta(hours=float(h))
    heights=z[t_idx,:,0,0]; temps=temp[t_idx,:,0,0]
    for hv,tv in zip(heights,temps):
        if np.ma.is_masked(hv) or np.ma.is_masked(tv):
            continue
        d=25.0-float(hv)
        if 0<=d<=25.0:
            recs.append({'datetime':dt,'depth':round(d),'temp_sim':float(tv)})
sim=pd.DataFrame(recs).groupby(['datetime','depth']).agg({'temp_sim':'mean'}).reset_index()
merged=pd.merge(obs,sim,on=['datetime','depth'],how='inner')
