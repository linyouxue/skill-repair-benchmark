import openpyxl
from openpyxl.utils import get_column_letter
path='gdp.xlsx'
wb=openpyxl.load_workbook(path)
ws=wb['Task']
def set_block(r_start, r_end, c_start=8, c_end=12):
    # columns H(8) to L(12)
    for r in range(r_start, r_end+1):
        for c in range(c_start, c_end+1):
            col=get_column_letter(c)
            # year header is row 9 in this template
            formula=(
                f"=INDEX(Data!$H$21:$M$40,"
                f"MATCH($D{r},Data!$B$21:$B$40,0),"
                f"MATCH({col}$9,Data!$H$4:$M$4,0))"
            )
            ws[f"{col}{r}"].value=formula
# Step 1 blocks
