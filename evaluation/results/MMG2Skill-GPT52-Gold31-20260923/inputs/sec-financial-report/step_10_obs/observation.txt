import pandas as pd
from rapidfuzz import process, fuzz
def best_matches(cover_path, query, topn=5):
    df=pd.read_csv(cover_path,sep='\t',usecols=['ACCESSION_NUMBER','FILINGMANAGER_NAME'],dtype=str)
    names=df['FILINGMANAGER_NAME'].fillna('')
    # build list for rapidfuzz with index mapping
    results=process.extract(query, names.tolist(), scorer=fuzz.WRatio, limit=topn)
    out=[]
    for name, score, idx in results:
        out.append((score, df.loc[idx,'ACCESSION_NUMBER'], name))
    return out
for q in ['2025-q2','2025-q3']:
    cp=f'/root/{q}/COVERPAGE.tsv'
    for query in ['renaissance technologies','berkshire hathaway']:
        print('\n',q,query)
        for score, acc, name in best_matches(cp, query):
            print(f'{score:6.2f}  {acc}  {name}')
