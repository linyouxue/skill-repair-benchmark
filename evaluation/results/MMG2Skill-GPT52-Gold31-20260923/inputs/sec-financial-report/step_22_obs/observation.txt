def load_values(path, acc):
    vals=defaultdict(int)
    for chunk in pd.read_csv(path, sep='\t', dtype={'ACCESSION_NUMBER':str,'CUSIP':str,'VALUE':str}, usecols=['ACCESSION_NUMBER','CUSIP','VALUE'], chunksize=1_000_000):
        sub=chunk[chunk['ACCESSION_NUMBER']==acc]
        if sub.empty:
            continue
        # VALUE numeric
        for cusip, v in zip(sub['CUSIP'], sub['VALUE']):
            if pd.isna(cusip) or pd.isna(v):
                continue
            try:
                vals[cusip] += int(float(v))
            except Exception:
                pass
    return vals
