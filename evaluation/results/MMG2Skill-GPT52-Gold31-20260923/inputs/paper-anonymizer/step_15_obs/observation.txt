import fitz
def find_pages(path, needles):
    doc=fitz.open(path)
    hits={n:[] for n in needles}
    for pno,page in enumerate(doc):
        t=page.get_text('text').lower()
        for n in needles:
            if n.lower() in t:
                hits[n].append(pno)
    doc.close()
    return {k:v for k,v in hits.items() if v}
