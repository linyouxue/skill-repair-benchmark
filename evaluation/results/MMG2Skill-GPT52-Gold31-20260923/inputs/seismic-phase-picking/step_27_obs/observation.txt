TARGET_STD=1e-6
def mkst(fn):
    z=np.load(fn, allow_pickle=True)
    x=z['data'].astype('float32')
    std=float(x.std())

        x=x*(TARGET_STD/std)
    dt=float(z['dt']); sr=1/dt

        tr.stats.location=str(z.get('location_code',''))
        tr.stats.channel=ch
        tr.stats.starttime=stime
        tr.stats.sampling_rate=sr
        st+=tr
    return st
