package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import clusterdata.utils.io.GzipCsvEventSource;
import clusterdata.utils.io.LocalTextFileSink;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.functions.windowing.ProcessWindowFunction;
import org.apache.flink.util.Collector;
import org.apache.flink.api.common.RuntimeExecutionMode;

import java.time.Duration;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // Session inactivity gap in seconds; default 600 == 10 minutes.
        final int sesssize = params.getInt("sesssize", 600);
        final long sessionGapMs = sesssize * 1000L;

        if (outputPath == null) {
            throw new IllegalArgumentException("Missing required parameter: --output");
        }
        if (taskInput == null) {
            throw new IllegalArgumentException("Missing required parameter: --task_input");
        }
        if (jobInput == null) {
            throw new IllegalArgumentException("Missing required parameter: --job_input");
        }

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setRuntimeMode(RuntimeExecutionMode.BATCH);
        env.setParallelism(1);

        WatermarkStrategy<TaskEvent> taskWm = WatermarkStrategy
                .<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(1))
                .withTimestampAssigner((e, ts) -> e.timestampUs / 1000L);

        WatermarkStrategy<JobEvent> jobWm = WatermarkStrategy
                .<JobEvent>forBoundedOutOfOrderness(Duration.ofSeconds(1))
                .withTimestampAssigner((e, ts) -> e.timestampUs / 1000L);

        DataStream<TaskEvent> taskEventsStream = env
                .addSource(taskSourceOrTest(new GzipCsvEventSource<>(taskInput, TaskEvent::fromCsvLine)))
                .name("task-events")
                .assignTimestampsAndWatermarks(taskWm);

        DataStream<JobEvent> jobEventsStream = env
                .addSource(jobSourceOrTest(new GzipCsvEventSource<>(jobInput, JobEvent::fromCsvLine)))
                .name("job-events")
                .assignTimestampsAndWatermarks(jobWm);

        // Stage == event-time session window over SUBMIT events keyed by jobId.
        SingleOutputStreamOperator<Tuple2<Long, Integer>> stageCounts = taskEventsStream
                .filter(e -> e.eventType == TaskEvent.EVENT_SUBMIT)
                .name("filter-submit")
                .keyBy(TaskEvent::getJobId)
                .window(EventTimeSessionWindows.withGap(Time.milliseconds(sessionGapMs)))
                .process(new CountInWindow())
                .name("stage-counts");

        // Per job, compute maximum stage size.
        SingleOutputStreamOperator<Tuple2<Long, Integer>> longestStagePerJob = stageCounts
                .keyBy(t -> t.f0)
                .reduce((a, b) -> Tuple2.of(a.f0, Math.max(a.f1, b.f1)))
                .name("max-stage-count");

        // Terminal job event (FINISH or KILL) per job.
        SingleOutputStreamOperator<Tuple2<Long, Long>> terminalJobSignals = jobEventsStream
                .filter(e -> e.eventType == JobEvent.EVENT_FINISH || e.eventType == JobEvent.EVENT_KILL)
                .name("filter-terminal")
                .map(e -> Tuple2.of(e.jobId, e.timestampUs / 1000L))
                .name("terminal-signal");

        SingleOutputStreamOperator<String> outputLines = longestStagePerJob
                .keyBy(t -> t.f0)
                .connect(terminalJobSignals.keyBy(t -> t.f0))
                .process(new EmitWhenTerminalSeen(sessionGapMs))
                .name("emit-after-terminal")
                .map(t -> String.format("(%d,%d)", t.f0, t.f1))
                .name("format-output");

        outputLines
                .addSink((org.apache.flink.streaming.api.functions.sink.SinkFunction<String>) sinkOrTest(new LocalTextFileSink(outputPath)))
                .name("local-file-sink")
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    private static final class CountInWindow extends ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow> {
        @Override
        public void process(Long key, Context context, Iterable<TaskEvent> elements, Collector<Tuple2<Long, Integer>> out) {
            int c = 0;
            for (TaskEvent ignored : elements) {
                c++;
            }
            out.collect(Tuple2.of(key, c));
        }
    }

    private static final class EmitWhenTerminalSeen
            extends KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, Tuple2<Long, Long>, Tuple2<Long, Integer>> {

        private final long sessionGapMs;

        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Long> terminalTsMs;
        private transient ValueState<Boolean> emitted;

        private EmitWhenTerminalSeen(long sessionGapMs) {
            this.sessionGapMs = sessionGapMs;
        }

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) {
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            terminalTsMs = getRuntimeContext().getState(new ValueStateDescriptor<>("terminalTsMs", Long.class));
            emitted = getRuntimeContext().getState(new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement1(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Boolean already = emitted.value();
            if (already != null && already) {
                return;
            }

            Integer current = maxStageCount.value();
            if (current == null || value.f1 > current) {
                maxStageCount.update(value.f1);
            }
        }

        @Override
        public void processElement2(Tuple2<Long, Long> terminal, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Boolean already = emitted.value();
            if (already != null && already) {
                return;
            }

            Long existing = terminalTsMs.value();
            if (existing == null) {
                terminalTsMs.update(terminal.f1);
                ctx.timerService().registerEventTimeTimer(terminal.f1 + sessionGapMs);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Boolean already = emitted.value();
            if (already != null && already) {
                return;
            }

            Integer max = maxStageCount.value();
            if (max == null) {
                max = 0;
            }
            out.collect(Tuple2.of(ctx.getCurrentKey(), max));
            emitted.update(true);
        }
    }
}
