from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from sympy import Rational, nsimplify


def _coord_to_fraction(value: float, max_denominator: int = 12) -> str:
    rational = nsimplify(float(value), rational=True, tolerance=1e-2)
    rational = Rational(rational).limit_denominator(max_denominator)
    if rational.q == 1:
        return str(rational.p)
    return f"{rational.p}/{rational.q}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    structure = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(structure)
    dataset = sga.get_symmetry_dataset()

    wyckoff_letters = list(dataset.wyckoffs)
    equivalent_atoms = list(dataset.equivalent_atoms)

    multiplicity: dict[str, int] = {}
    first_index: dict[str, int] = {}
    for site_idx, letter in enumerate(wyckoff_letters):
        multiplicity[letter] = multiplicity.get(letter, 0) + 1
        group_repr = equivalent_atoms[site_idx]
        if letter not in first_index or group_repr < first_index[letter]:
            first_index[letter] = group_repr

    coordinates: dict[str, list[str]] = {}
    for letter, idx in first_index.items():
        frac = structure[idx].frac_coords
        coordinates[letter] = [_coord_to_fraction(c) for c in frac]

    ordered_letters = sorted(multiplicity.keys())
    return {
        "wyckoff_multiplicity_dict": {l: multiplicity[l] for l in ordered_letters},
        "wyckoff_coordinates_dict": {l: coordinates[l] for l in ordered_letters},
    }


if __name__ == "__main__":
