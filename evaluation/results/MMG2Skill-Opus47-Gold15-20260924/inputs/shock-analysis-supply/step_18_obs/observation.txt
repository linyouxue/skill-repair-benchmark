        r = urllib.request.urlopen(req, timeout=30)
        html = r.read().decode('utf-8', errors='ignore')
        # find file links
        import re
        links = re.findall(r'href="([^"]*pwt[^"]*\.(?:xlsx?|dta|csv))"', html, re.IGNORECASE)
        print(u, "->", set(links))
    except Exception as e:
        print(u, "ERR", e)
