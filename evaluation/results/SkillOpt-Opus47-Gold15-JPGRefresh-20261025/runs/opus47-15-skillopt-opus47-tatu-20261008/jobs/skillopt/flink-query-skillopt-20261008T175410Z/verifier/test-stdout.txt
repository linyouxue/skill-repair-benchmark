/**
 * For each finished job, output (jobId, N) where N is the number of task-SUBMIT
 * events in the busiest "stage" of that job. A stage is an event-time session
 * window of task SUBMIT events (event_type == 0) with a 10-minute inactivity gap.
 * A job is "finished" when a FINISH (event_type == 4) job event is observed.
 */
public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        // Session inactivity gap in seconds (default: 10 minutes).
        final int sesssize = params.getInt("sesssize", 600);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        WatermarkStrategy<TaskEvent> taskWm = WatermarkStrategy
                .<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(5))
                .withTimestampAssigner((e, ts) -> e.getEventTimeMillis());

        WatermarkStrategy<JobEvent> jobWm = WatermarkStrategy
                .<JobEvent>forBoundedOutOfOrderness(Duration.ofSeconds(5))
                .withTimestampAssigner((e, ts) -> e.getEventTimeMillis());

        DataStream<TaskEvent> taskEvents =
                env.addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                   .returns(TaskEvent.class)
                   .assignTimestampsAndWatermarks(taskWm);

        DataStream<JobEvent> jobEvents =
                env.addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                   .returns(JobEvent.class)
                   .assignTimestampsAndWatermarks(jobWm);

        // Count SUBMIT (event_type == 0) events per job per 10-min session.
        // Every SUBMIT is counted — resubmissions of the same task_index count again.
        SingleOutputStreamOperator<Tuple2<Long, Integer>> perSessionCounts = taskEvents
                .filter((FilterFunction<TaskEvent>) e -> e.eventType == 0)
                .keyBy((KeySelector<TaskEvent, Long>) e -> e.jobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .process(new CountSubmitsPerSession());

        // Reduce per-job to its longest (max-count) stage.
        SingleOutputStreamOperator<Tuple2<Long, Integer>> maxPerJob = perSessionCounts
                .keyBy((KeySelector<Tuple2<Long, Integer>, Long>) t -> t.f0)
                .maxBy(1);

        // Collect job IDs that FINISH (event_type == 4).
        SingleOutputStreamOperator<Long> finishedJobs = jobEvents
                .filter((FilterFunction<JobEvent>) e -> e.eventType == 4)
                .map((MapFunction<JobEvent, Long>) e -> e.jobId)
                .returns(Long.class);

        KeyedStream<Tuple2<Long, Integer>, Long> keyedMax =
                maxPerJob.keyBy((KeySelector<Tuple2<Long, Integer>, Long>) t -> t.f0);
        KeyedStream<Long, Long> keyedFinished =
                finishedJobs.keyBy((KeySelector<Long, Long>) j -> j);

        DataStream<String> results = keyedMax
                .connect(keyedFinished)
                .process(new JoinMaxAndFinished());

        if (outputPath != null) {
            results.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);
        } else {
            results.print().setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

    /** Count every SUBMIT event in the session window. */
    public static class CountSubmitsPerSession
            extends ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow> {
        @Override
        public void process(Long jobId, Context ctx, Iterable<TaskEvent> elements,
                            Collector<Tuple2<Long, Integer>> out) {
            int count = 0;
            for (TaskEvent ignored : elements) {
                count++;
            }
            out.collect(Tuple2.of(jobId, count));
        }
    }

    /**
     * Joins the per-job max-stage stream with the finished-jobs stream.
     * Emits (jobId, maxCount) only after BOTH sides have been seen for a jobId.
     * Also registers an end-of-time timer so pending state flushes on bounded input.
     */
    public static class JoinMaxAndFinished
            extends KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, Long, String> {

        private transient ValueState<Integer> maxState;
        private transient ValueState<Boolean> finishedState;
        private transient ValueState<Boolean> emittedState;

        @Override
        public void open(Configuration parameters) {
            maxState = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("max", Integer.class));
            finishedState = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("finished", Boolean.class));
            emittedState = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement1(Tuple2<Long, Integer> value, Context ctx,
                                    Collector<String> out) throws Exception {
            maxState.update(value.f1);
            tryEmit(ctx, out);
            ctx.timerService().registerEventTimeTimer(Long.MAX_VALUE - 1);
        }

        @Override
        public void processElement2(Long jobId, Context ctx,
                                    Collector<String> out) throws Exception {
            finishedState.update(Boolean.TRUE);
            tryEmit(ctx, out);
            ctx.timerService().registerEventTimeTimer(Long.MAX_VALUE - 1);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx,
                            Collector<String> out) throws Exception {
            tryEmit(ctx, out);
        }

        private void tryEmit(KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, Long, String>.Context ctx,
                             Collector<String> out) throws Exception {
            Boolean emitted = emittedState.value();
            if (emitted != null && emitted) return;
            Integer max = maxState.value();
            Boolean finished = finishedState.value();
            if (max != null && finished != null && finished) {
                out.collect("(" + ctx.getCurrentKey() + "," + max + ")");
                emittedState.update(Boolean.TRUE);
            }
        }
    }
