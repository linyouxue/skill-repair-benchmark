package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import clusterdata.utils.JobEventSource;
import clusterdata.utils.LocalFileSink;
import clusterdata.utils.TaskEventSource;
import org.apache.flink.api.common.RuntimeExecutionMode;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.common.typeinfo.Types;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.windowing.ProcessWindowFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;


public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // Session gap in seconds. Default 600s = 10 minutes.
        final int sesssize = params.getInt("sesssize", 600);
        final long gapMillis = sesssize * 1_000L;

        if (taskInput == null || jobInput == null || outputPath == null) {
            throw new IllegalArgumentException("Required parameters: --task_input, --job_input, --output");
        }

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setRuntimeMode(RuntimeExecutionMode.BATCH);

        DataStream<TaskEvent> tasks = env
                .addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((e, ts) -> e.getTimestampMillis()));

        DataStream<JobEvent> jobEvents = env
                .addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((e, ts) -> e.getTimestampMillis()));

        // Count SUBMIT events per job in event-time session windows ("stages").
        DataStream<Tuple2<Long, Integer>> stageCounts = tasks
                .filter(e -> e.getEventType() == TaskEvent.EVENT_SUBMIT)
                .keyBy(TaskEvent::getJobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .process(new ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void process(Long jobId, Context context, Iterable<TaskEvent> elements, Collector<Tuple2<Long, Integer>> out) {
                        int cnt = 0;
                        for (TaskEvent ignored : elements) {
                            cnt++;
                        }
                        out.collect(Tuple2.of(jobId, cnt));
                    }
                });

        DataStream<JobEvent> finishes = jobEvents.filter(e -> e.getEventType() == JobEvent.EVENT_FINISH);

        DataStream<String> results = stageCounts
                .keyBy(t -> t.f0)
                .connect(finishes.keyBy(JobEvent::getJobId))
                .process(new KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, JobEvent, String>() {

                    private transient ValueState<Integer> maxStage;
                    private transient ValueState<Boolean> finished;
                    private transient ValueState<Long> emitTimer;

                    @Override
                    public void open(Configuration parameters) {
                        maxStage = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("maxStage", Types.INT));
                        finished = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("finished", Types.BOOLEAN));
                        emitTimer = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("emitTimer", Types.LONG));
                    }

                    @Override
                    public void processElement1(Tuple2<Long, Integer> stageCount, Context ctx, Collector<String> out) throws Exception {
                        Integer currentMax = maxStage.value();
                        if (currentMax == null || stageCount.f1 > currentMax) {
                            maxStage.update(stageCount.f1);
                        }
                    }

                    @Override
                    public void processElement2(JobEvent finishEvent, Context ctx, Collector<String> out) throws Exception {
                        Boolean alreadyFinished = finished.value();
                        if (alreadyFinished != null && alreadyFinished) {
                            return;
                        }

                        finished.update(true);
                        long timerTs = finishEvent.getTimestampMillis() + gapMillis + 1L;
                        Long existingTimer = emitTimer.value();
                        if (existingTimer == null || timerTs > existingTimer) {
                            if (existingTimer != null) {
                                ctx.timerService().deleteEventTimeTimer(existingTimer);
                            }
                            emitTimer.update(timerTs);
                            ctx.timerService().registerEventTimeTimer(timerTs);
                        }
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
                        Boolean isFinished = finished.value();
                        if (isFinished == null || !isFinished) {
                            return;
                        }

                        Integer max = maxStage.value();
                        int result = (max == null) ? 0 : max;
                        out.collect("(" + ctx.getCurrentKey() + "," + result + ")");

                        maxStage.clear();
                        finished.clear();
                        emitTimer.clear();
                    }
                });

        results
                .addSink((org.apache.flink.streaming.api.functions.sink.SinkFunction<String>) sinkOrTest(new LocalFileSink(outputPath)))
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

}
