package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;

import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FlatMapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.common.typeinfo.TypeHint;
import org.apache.flink.api.common.typeinfo.TypeInformation;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.util.Collector;

import java.time.Duration;

/**
 * Longest "stage" (= run of task SUBMIT events separated by a short inactivity
 * gap) per Google-cluster-trace job.
 *
 * Schema notes grounded in /app/workspace/data/format.pdf:
 *   - Timestamps are in microseconds; divided by 1000 for Flink event time.
 *   - Task-event SUBMIT code = 0 (used to extend / open stages).
 *   - Job terminal event codes: EVICT=2, FAIL=3, FINISH=4, KILL=5, LOST=6.
 *     Any of these finalizes the job and triggers emission.
 *   - Every SUBMIT event counts, including resubmissions of the same
 *     (jobId, taskIndex), per the task description.
 *   - Inactivity gap: 10 minutes = 600_000 ms, strict ">" (a new stage opens
 *     only after more than 10 minutes without a SUBMIT for that job).
 *
 * Output contract: one line per job, formatted as "(jobId,longestStageTasks)".
 */
public class LongestSessionPerJob extends AppBase {

    private static final long GAP_MS = 10L * 60L * 1000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        System.out.println("output  " + outputPath);

        final StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        // Task-event SUBMITs -> (jobId, eventTimeMs, isTerminal=false).
        DataStream<Tuple2<Long, Long>> submits = env
                .readTextFile(taskInput)
                .flatMap(new FlatMapFunction<String, Tuple2<Long, Long>>() {
                    @Override
                    public void flatMap(String line, Collector<Tuple2<Long, Long>> out) {
                        TaskEvent e = TaskEvent.fromCsv(line);
                        if (e != null && e.eventType == TaskEvent.SUBMIT) {
                            out.collect(Tuple2.of(e.jobId, e.timestamp / 1000L));
                        }
                    }
                })
                .returns(TypeInformation.of(new TypeHint<Tuple2<Long, Long>>() {}));

        // Job terminal events -> (jobId, eventTimeMs).
        DataStream<Tuple2<Long, Long>> jobEnds = env
                .readTextFile(jobInput)
                .flatMap(new FlatMapFunction<String, Tuple2<Long, Long>>() {
                    @Override
                    public void flatMap(String line, Collector<Tuple2<Long, Long>> out) {
                        JobEvent e = JobEvent.fromCsv(line);
                        if (e != null && e.isTerminal()) {
                            out.collect(Tuple2.of(e.jobId, e.timestamp / 1000L));
                        }
                    }
                })
                .returns(TypeInformation.of(new TypeHint<Tuple2<Long, Long>>() {}));

        // Tag with a flag: 0 = SUBMIT, 1 = JOB_END. Keep ts for event time.
        // Shape: (jobId, ts, kind).
        TypeInformation<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>> tripleType =
                TypeInformation.of(new TypeHint<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>>() {});

        DataStream<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>> submitTriples =
                submits.map(t -> org.apache.flink.api.java.tuple.Tuple3.of(t.f0, t.f1, 0))
                        .returns(tripleType);
        DataStream<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>> endTriples =
                jobEnds.map(t -> org.apache.flink.api.java.tuple.Tuple3.of(t.f0, t.f1, 1))
                        .returns(tripleType);

        DataStream<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>> unioned =
                submitTriples.union(endTriples);

        WatermarkStrategy<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>> ws =
                WatermarkStrategy
                        .<org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>>forBoundedOutOfOrderness(
                                Duration.ofMinutes(1))
                        .withTimestampAssigner((event, ts) -> event.f1);

        DataStream<String> results = unioned
                .assignTimestampsAndWatermarks(ws)
                .keyBy(t -> t.f0)
                .process(new SessionCounter());

        results.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    /**
     * Per-job state machine.
     *
     * SUBMIT events arrive via one stream, terminal job events via another;
     * union merges them with min-of-watermarks, but inside the process function
     * elements still arrive in arrival order (which may interleave the two
     * sources). To make emission deterministic, a terminal event only records
     * terminalTs in state and sets an event-time timer at terminalTs. By the
     * time the timer fires, every SUBMIT with ts <= terminalTs has been
     * processed (bounded-out-of-orderness watermarks guarantee it), so the
     * accumulated bestStageCount is final.
     */
    public static class SessionCounter
            extends KeyedProcessFunction<Long, org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer>, String> {

        private transient ValueState<Long> lastSubmitTs;
        private transient ValueState<Long> currentStageCount;
        private transient ValueState<Long> bestStageCount;
        private transient ValueState<Long> terminalTs;

        @Override
        public void open(Configuration parameters) {
            lastSubmitTs = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("lastSubmitTs", Long.class));
            currentStageCount = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("currentStageCount", Long.class));
            bestStageCount = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("bestStageCount", Long.class));
            terminalTs = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("terminalTs", Long.class));
        }

        @Override
        public void processElement(
                org.apache.flink.api.java.tuple.Tuple3<Long, Long, Integer> value,
                Context ctx,
                Collector<String> out) throws Exception {

            long ts = value.f1;
            int kind = value.f2;

            if (kind == 0) {
                Long last = lastSubmitTs.value();
                Long curr = currentStageCount.value();
                Long best = bestStageCount.value();
                if (best == null) best = 0L;

                if (last == null || ts - last > GAP_MS) {
                    if (curr != null && curr > best) best = curr;
                    curr = 1L;
                } else {
                    curr = (curr == null ? 1L : curr + 1L);
                }
                lastSubmitTs.update(ts);
                currentStageCount.update(curr);
                bestStageCount.update(best);
            } else {
                // Defer terminal emission until watermark passes terminalTs
                // so any late SUBMITs with ts <= terminalTs are counted first.
                Long existing = terminalTs.value();
                long t = (existing == null) ? ts : Math.min(existing, ts);
                terminalTs.update(t);
                ctx.timerService().registerEventTimeTimer(t);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            Long t = terminalTs.value();
            if (t == null || t.longValue() != timestamp) {
                return;
            }
            long jobId = ctx.getCurrentKey();
            Long curr = currentStageCount.value();
            Long best = bestStageCount.value();
            if (best == null) best = 0L;
            if (curr != null && curr > best) best = curr;
            out.collect("(" + jobId + "," + best + ")");
            lastSubmitTs.clear();
            currentStageCount.clear();
            bestStageCount.clear();
            terminalTs.clear();
        }
    }
}
