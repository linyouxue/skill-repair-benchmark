package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.SerializableTimestampAssigner;

import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.util.Collector;

import java.time.Duration;

public class LongestSessionPerJob extends AppBase {

    private static final int TASK_SUBMIT = 0;
    private static final long MICROS_PER_SECOND = 1_000_000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final long sessionGapMicros = params.getInt("sesssize", 600) * MICROS_PER_SECOND;

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        WatermarkStrategy<TaskEvent> taskWatermarks = WatermarkStrategy
                .<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                .withTimestampAssigner((SerializableTimestampAssigner<TaskEvent>) (event, timestamp) -> event.timestamp);

        WatermarkStrategy<JobEvent> jobWatermarks = WatermarkStrategy
                .<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                .withTimestampAssigner((SerializableTimestampAssigner<JobEvent>) (event, timestamp) -> event.timestamp);

        DataStream<TaskEvent> taskEvents = env
                .readTextFile(taskInput)
                .map(TaskEvent::fromCsvLine)
                .filter(event -> event.eventType == TASK_SUBMIT)
                .assignTimestampsAndWatermarks(taskWatermarks);

        DataStream<JobEvent> jobEvents = env
                .readTextFile(jobInput)
                .map(JobEvent::fromCsvLine)
                .filter(event -> event.eventType >= 3)
                .assignTimestampsAndWatermarks(jobWatermarks);

        DataStream<Tuple2<Long, Integer>> results = taskEvents
                .keyBy(event -> event.jobId)
                .connect(jobEvents.keyBy(event -> event.jobId))
                .process(new LongestStageFunction(sessionGapMicros));

        results
                .map(result -> "(" + result.f0 + "," + result.f1 + ")")
                .writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE);

        env.execute("LongestSessionPerJob");
    }

    private static class LongestStageFunction extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {

        private final long sessionGapMicros;

        private transient ValueState<Long> lastSubmitTs;
        private transient ValueState<Integer> currentStageCount;
        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Long> terminalTs;
        private transient ValueState<Boolean> emitted;

        private LongestStageFunction(long sessionGapMicros) {
            this.sessionGapMicros = sessionGapMicros;
        }

        @Override
        public void open(Configuration parameters) {
            lastSubmitTs = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTs", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            terminalTs = getRuntimeContext().getState(new ValueStateDescriptor<>("terminalTs", Long.class));
            emitted = getRuntimeContext().getState(new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Long lastTs = lastSubmitTs.value();
            Integer currentCount = defaultInt(currentStageCount.value());
            Integer maxCount = defaultInt(maxStageCount.value());
            Long terminal = terminalTs.value();

            if (terminal != null && event.timestamp > terminal) {
                return;
            }

            if (lastTs != null && event.timestamp - lastTs >= sessionGapMicros) {
                maxCount = Math.max(maxCount, currentCount);
                currentCount = 0;
            }

            currentCount++;
            currentStageCount.update(currentCount);
            maxStageCount.update(Math.max(maxCount, currentCount));
            lastSubmitTs.update(event.timestamp);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Long existingTerminal = terminalTs.value();
            if (existingTerminal == null || event.timestamp < existingTerminal) {
                terminalTs.update(event.timestamp);
                ctx.timerService().registerEventTimeTimer(event.timestamp + 1);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            if (Boolean.TRUE.equals(emitted.value())) {
                return;
            }

            Integer currentCount = defaultInt(currentStageCount.value());
            Integer maxCount = Math.max(defaultInt(maxStageCount.value()), currentCount);

            if (maxCount > 0) {
                out.collect(Tuple2.of(ctx.getCurrentKey(), maxCount));
            }

            emitted.update(true);
            clearState();
        }

        private void clearState() throws Exception {
            lastSubmitTs.clear();
            currentStageCount.clear();
            maxStageCount.clear();
            terminalTs.clear();
            emitted.clear();
        }

        private int defaultInt(Integer value) {
            return value == null ? 0 : value;
        }
    }
}
