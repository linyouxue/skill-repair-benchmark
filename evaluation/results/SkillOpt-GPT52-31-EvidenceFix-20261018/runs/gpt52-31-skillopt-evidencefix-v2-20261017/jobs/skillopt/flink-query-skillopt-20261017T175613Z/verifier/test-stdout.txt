package clusterdata.query;

import java.time.Duration;

public class LongestSessionPerJob extends AppBase {

    private static final long SESSION_GAP_MICROS = 10L * 60L * 1_000_000L;
    private static final long ALLOWED_LATENESS_MICROS = 60L * 1_000_000L;
    private static final int TASK_SUBMIT = 0;
    private static final int JOB_FINISH = 4;
    private static final int JOB_KILL = 5;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        WatermarkStrategy<TaskEvent> taskWatermarks = WatermarkStrategy
                .<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                .withTimestampAssigner(new SerializableTimestampAssigner<TaskEvent>() {
                    @Override
                    public long extractTimestamp(TaskEvent element, long recordTimestamp) {
                        return element.timestamp;
                    }
                });

        WatermarkStrategy<JobEvent> jobWatermarks = WatermarkStrategy
                .<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                .withTimestampAssigner(new SerializableTimestampAssigner<JobEvent>() {
                    @Override
                    public long extractTimestamp(JobEvent element, long recordTimestamp) {
                        return element.timestamp;
                    }
                });

        DataStream<TaskEvent> taskStream;
        if (taskEvents == null) {
            taskStream = env
                    .readTextFile(taskInput)
                    .map(TaskEvent::fromCsv);
        } else {
            taskStream = env.addSource(taskEvents);
        }

        taskStream = taskStream
                .filter(event -> event.eventType == TASK_SUBMIT)
                .assignTimestampsAndWatermarks(taskWatermarks);

        DataStream<JobEvent> jobStream;
        if (jobEvents == null) {
            jobStream = env
                    .readTextFile(jobInput)
                    .map(JobEvent::fromCsv);
        } else {
            jobStream = env.addSource(jobEvents);
        }

        jobStream = jobStream
                .filter(event -> event.eventType == JOB_FINISH || event.eventType == JOB_KILL)
                .assignTimestampsAndWatermarks(jobWatermarks);

        DataStream<Tuple2<Long, Integer>> result = taskStream
                .keyBy(event -> event.jobId)
                .connect(jobStream.keyBy(event -> event.jobId))
                .process(new LongestStageProcessFunction());

        result
                .map(value -> "(" + value.f0 + "," + value.f1 + ")")
                .writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    public static class LongestStageProcessFunction extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {

        private transient ValueState<Long> lastSubmitTs;
        private transient ValueState<Integer> currentStageCount;
        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Long> terminalTs;
        private transient ValueState<Boolean> emitted;

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) {
            lastSubmitTs = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTs", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            terminalTs = getRuntimeContext().getState(new ValueStateDescriptor<>("terminalTs", Long.class));
            emitted = getRuntimeContext().getState(new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent event, Context context, Collector<Tuple2<Long, Integer>> collector) throws Exception {
            Long lastTs = lastSubmitTs.value();
            Integer currentCount = currentStageCount.value();
            Integer maxCount = maxStageCount.value();
            Long finishTs = terminalTs.value();

            if (finishTs != null && event.timestamp > finishTs) {
                return;
            }

            if (currentCount == null) {
                currentCount = 0;
            }

            if (maxCount == null) {
                maxCount = 0;
            }

            if (lastTs != null && event.timestamp - lastTs >= SESSION_GAP_MICROS) {
                maxCount = Math.max(maxCount, currentCount);
                currentCount = 0;
            }

            currentCount += 1;

            lastSubmitTs.update(event.timestamp);
            currentStageCount.update(currentCount);
            maxStageCount.update(maxCount);
        }

        @Override
        public void processElement2(JobEvent event, Context context, Collector<Tuple2<Long, Integer>> collector) throws Exception {
            Long existingTerminalTs = terminalTs.value();
            long effectiveTerminalTs = existingTerminalTs == null
                    ? event.timestamp
                    : Math.min(existingTerminalTs, event.timestamp);

            terminalTs.update(effectiveTerminalTs);
            context.timerService().registerEventTimeTimer(effectiveTerminalTs + ALLOWED_LATENESS_MICROS);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Boolean alreadyEmitted = emitted.value();
            if (alreadyEmitted != null && alreadyEmitted) {
                return;
            }

            Integer currentCount = currentStageCount.value();
            Integer maxCount = maxStageCount.value();
            Long finishTs = terminalTs.value();

            if (finishTs == null || currentCount == null) {
                clearState();
                return;
            }

            if (maxCount == null) {
                maxCount = 0;
            }

            maxCount = Math.max(maxCount, currentCount);
            out.collect(Tuple2.of(ctx.getCurrentKey(), maxCount));

            emitted.update(true);
            clearState();
        }

        private void clearState() throws Exception {
            lastSubmitTs.clear();
            currentStageCount.clear();
            maxStageCount.clear();
            terminalTs.clear();
            emitted.clear();
        }
    }
