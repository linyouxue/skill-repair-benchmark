package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;

public class LongestSessionPerJob extends AppBase {

    private static final int TASK_SUBMIT_EVENT_TYPE = 0;

    private static boolean isTerminalJobEvent(int eventType) {
        // ClusterData job_event types are small ints; terminal events include FAIL/FINISH/KILL/LOST.
        return eventType == 3 || eventType == 4 || eventType == 5 || eventType == 6;
    }

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssizeSeconds = params.getInt("sesssize", 600);

        if (taskInput == null || jobInput == null || outputPath == null) {
            throw new IllegalArgumentException("Required params: --task_input --job_input --output");
        }

        final long sessionGapMs = sesssizeSeconds * 1000L;

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        SourceFunction<TaskEvent> taskSource = taskSourceOrTest(new GzipCsvSourceFunction<>(taskInput, TaskEvent::fromCsv));
        SourceFunction<JobEvent> jobSource = jobSourceOrTest(new GzipCsvSourceFunction<>(jobInput, JobEvent::fromCsv));

        WatermarkStrategy<TaskEvent> taskWm = WatermarkStrategy
                .<TaskEvent>forBoundedOutOfOrderness(Duration.ZERO)
                .withTimestampAssigner((e, ts) -> e.timestampMicros / 1000L);

        WatermarkStrategy<JobEvent> jobWm = WatermarkStrategy
                .<JobEvent>forBoundedOutOfOrderness(Duration.ZERO)
                .withTimestampAssigner((e, ts) -> e.timestampMicros / 1000L);

        KeyedStream<TaskEvent, Long> tasks = env
                .addSource(taskSource)
                .assignTimestampsAndWatermarks(taskWm)
                .keyBy(e -> e.jobId);

        KeyedStream<JobEvent, Long> jobs = env
                .addSource(jobSource)
                .assignTimestampsAndWatermarks(jobWm)
                .keyBy(e -> e.jobId);

        DataStream<Tuple2<Long, Long>> results = tasks
                .connect(jobs)
                .process(new LongestStageOnJobFinish(sessionGapMs));

        @SuppressWarnings("unchecked")
        org.apache.flink.streaming.api.functions.sink.SinkFunction<Tuple2<Long, Long>> sink =
                (org.apache.flink.streaming.api.functions.sink.SinkFunction<Tuple2<Long, Long>>) sinkOrTest(new LocalFileSink(outputPath));

        results
                .addSink(sink)
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    private interface LineParser<T> {
        T parse(String line);
    }

    private static final class GzipCsvSourceFunction<T> extends RichSourceFunction<T> {
        private final String path;
        private final LineParser<T> parser;
        private volatile boolean running = true;

        private GzipCsvSourceFunction(String path, LineParser<T> parser) {
            this.path = path;
            this.parser = parser;
        }

        @Override
        public void run(SourceContext<T> ctx) throws Exception {
            try (BufferedReader reader = new BufferedReader(new InputStreamReader(
                    new GZIPInputStream(new FileInputStream(path)), StandardCharsets.UTF_8))) {
                String line;
                while (running && (line = reader.readLine()) != null) {
                    if (line.isEmpty()) {
                        continue;
                    }
                    ctx.collect(parser.parse(line));
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    private static final class LocalFileSink extends RichSinkFunction<Tuple2<Long, Long>> {
        private final String path;
        private transient BufferedWriter writer;

        private LocalFileSink(String path) {
            this.path = path;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            File f = new File(path);
            File parent = f.getParentFile();
            if (parent != null && !parent.exists()) {
                parent.mkdirs();
            }
            writer = new BufferedWriter(new FileWriter(f, false));
        }

        @Override
        public void invoke(Tuple2<Long, Long> value, Context context) throws Exception {
            writer.write(value.toString());
            writer.newLine();
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.flush();
                writer.close();
            }
        }
    }

    private static final class LongestStageOnJobFinish extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Long>> {

        private final long sessionGapMs;

        private transient ValueState<Long> lastSubmitMs;
        private transient ValueState<Long> currentStageCount;
        private transient ValueState<Long> maxStageCount;
        private transient ValueState<Boolean> hasQualifyingActivity;
        private transient ValueState<Long> terminalJobTsMs;
        private transient ValueState<Boolean> emitted;

        private LongestStageOnJobFinish(long sessionGapMs) {
            this.sessionGapMs = sessionGapMs;
        }

        @Override
        public void open(Configuration parameters) {
            lastSubmitMs = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitMs", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Long.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Long.class));
            hasQualifyingActivity = getRuntimeContext().getState(new ValueStateDescriptor<>("hasQualifyingActivity", Boolean.class));
            terminalJobTsMs = getRuntimeContext().getState(new ValueStateDescriptor<>("terminalJobTsMs", Long.class));
            emitted = getRuntimeContext().getState(new ValueStateDescriptor<>("emitted", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent value, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (value.eventType != TASK_SUBMIT_EVENT_TYPE) {
                return;
            }

            hasQualifyingActivity.update(true);

            long tsMs = value.timestampMicros / 1000L;
            Long last = lastSubmitMs.value();
            Long cur = currentStageCount.value();
            Long max = maxStageCount.value();

            if (cur == null) {
                cur = 0L;
            }
            if (max == null) {
                max = 0L;
            }

            if (last == null || last < 0L || tsMs - last >= sessionGapMs) {
                max = Math.max(max, cur);
                cur = 1L;
            } else {
                cur = cur + 1L;
            }

            lastSubmitMs.update(tsMs);
            currentStageCount.update(cur);
            maxStageCount.update(max);
        }

        @Override
        public void processElement2(JobEvent value, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (!isTerminalJobEvent(value.eventType)) {
                return;
            }

            long tsMs = value.timestampMicros / 1000L;
            Long prevTerminal = terminalJobTsMs.value();
            if (prevTerminal == null || tsMs > prevTerminal) {
                terminalJobTsMs.update(tsMs);
                ctx.timerService().registerEventTimeTimer(tsMs);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (Boolean.TRUE.equals(emitted.value())) {
                return;
            }
            if (!Boolean.TRUE.equals(hasQualifyingActivity.value())) {
                return;
            }
            Long terminal = terminalJobTsMs.value();
            if (terminal == null || timestamp != terminal) {
                return;
            }

            Long cur = currentStageCount.value();
            Long max = maxStageCount.value();
            if (cur == null) {
                cur = 0L;
            }
            if (max == null) {
                max = 0L;
            }
            long result = Math.max(max, cur);
            if (result <= 0L) {
                return;
            }

            out.collect(Tuple2.of(ctx.getCurrentKey(), result));
            emitted.update(true);
        }
    }
