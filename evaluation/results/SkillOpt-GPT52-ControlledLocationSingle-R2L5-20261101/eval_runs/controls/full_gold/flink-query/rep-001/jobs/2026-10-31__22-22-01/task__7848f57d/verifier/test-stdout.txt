public class LongestSessionPerJob extends AppBase {

    private static final int TASK_EVENT_SUBMIT = 0;
    private static final int JOB_EVENT_FINISH = 4;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        final int sesssizeSeconds = params.getInt("sesssize", 600);

        if (taskInput == null || jobInput == null || outputPath == null) {
            throw new IllegalArgumentException("Missing required parameters: task_input, job_input, output");
        }

        final long sessionGapMicros = sesssizeSeconds * 1_000_000L;

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        DataStream<TaskEvent> taskEvents;
        if (AppBase.taskEvents == null) {
            taskEvents = env
                    .readTextFile(taskInput)
                    .name("task-event-lines")
                    .map(new ParseTaskEvent());
        } else {
            taskEvents = env.addSource(AppBase.taskEvents);
        }

        taskEvents = taskEvents.assignTimestampsAndWatermarks(
                        WatermarkStrategy
                                .<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(5))
                                .withTimestampAssigner(new SerializableTimestampAssigner<TaskEvent>() {
                                    @Override
                                    public long extractTimestamp(TaskEvent element, long recordTimestamp) {
                                        return element.timestamp;
                                    }
                                }))
                .name("task-event-watermarks");

        DataStream<JobEvent> jobEvents;
        if (AppBase.jobEvents == null) {
            jobEvents = env
                    .readTextFile(jobInput)
                    .name("job-event-lines")
                    .map(new ParseJobEvent());
        } else {
            jobEvents = env.addSource(AppBase.jobEvents);
        }

        jobEvents = jobEvents.assignTimestampsAndWatermarks(
                        WatermarkStrategy
                                .<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(5))
                                .withTimestampAssigner(new SerializableTimestampAssigner<JobEvent>() {
                                    @Override
                                    public long extractTimestamp(JobEvent element, long recordTimestamp) {
                                        return element.timestamp;
                                    }
                                }))
                .name("job-event-watermarks");

        SingleOutputStreamOperator<Tuple2<Long, Long>> longestStagePerJob = taskEvents
                .keyBy(e -> e.jobId)
                .connect(jobEvents.keyBy(e -> e.jobId))
                .process(new LongestStageOnFinish(sessionGapMicros))
                .name("longest-stage-per-job");

        SinkFunction<Tuple2<Long, Long>> sink = new LocalTupleFileSink(outputPath);

        longestStagePerJob.addSink((SinkFunction<Tuple2<Long, Long>>) sinkOrTest(sink)).name("file-sink");

        env.execute("LongestSessionPerJob");
    }


    private static class LocalTupleFileSink extends RichSinkFunction<Tuple2<Long, Long>> {
        private final String outputPath;
        private transient BufferedWriter writer;

        private LocalTupleFileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            File outFile = new File(outputPath);
            File parent = outFile.getParentFile();
            if (parent != null) {
                parent.mkdirs();
            }
            writer = new BufferedWriter(new OutputStreamWriter(new FileOutputStream(outFile, false), StandardCharsets.UTF_8));
        }

        @Override
        public void invoke(Tuple2<Long, Long> value, Context context) throws Exception {
            writer.write("(" + value.f0 + "," + value.f1 + ")");
            writer.newLine();
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.flush();
                writer.close();
            }
        }
    }

    private static class ParseTaskEvent implements MapFunction<String, TaskEvent> {
        @Override
        public TaskEvent map(String value) {
            String[] parts = value.split(",", -1);
            long ts = Long.parseLong(parts[0]);
            long jobId = Long.parseLong(parts[2]);
            int eventType = Integer.parseInt(parts[5]);
            return new TaskEvent(ts, jobId, eventType);
        }
    }

    private static class ParseJobEvent implements MapFunction<String, JobEvent> {
        @Override
        public JobEvent map(String value) {
            String[] parts = value.split(",", -1);
            long ts = Long.parseLong(parts[0]);
            long jobId = Long.parseLong(parts[2]);
            int eventType = Integer.parseInt(parts[3]);
            return new JobEvent(ts, jobId, eventType);
        }
    }

    private static class LongestStageOnFinish extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Long>> {
        private final long sessionGapMicros;

        private transient ValueState<Long> lastSubmitTimestamp;
        private transient ValueState<Long> currentStageCount;
        private transient ValueState<Long> maxStageCount;
        private transient ValueState<Boolean> hasQualifyingSubmit;

        private LongestStageOnFinish(long sessionGapMicros) {
            this.sessionGapMicros = sessionGapMicros;
        }

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) {
            lastSubmitTimestamp = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTimestamp", Long.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Long.class));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Long.class));
            hasQualifyingSubmit = getRuntimeContext().getState(new ValueStateDescriptor<>("hasQualifyingSubmit", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent value, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (value.eventType != TASK_EVENT_SUBMIT) {
                return;
            }

            Long lastTs = lastSubmitTimestamp.value();
            Long cur = currentStageCount.value();
            if (cur == null) {
                cur = 0L;
            }

            if (lastTs == null || value.timestamp - lastTs > sessionGapMicros) {
                cur = 1L;
            } else {
                cur = cur + 1L;
            }

            currentStageCount.update(cur);
            lastSubmitTimestamp.update(value.timestamp);

            Long max = maxStageCount.value();
            if (max == null || cur > max) {
                maxStageCount.update(cur);
            }

            hasQualifyingSubmit.update(true);
        }

        @Override
        public void processElement2(JobEvent value, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (value.eventType != JOB_EVENT_FINISH) {
                return;
            }

            if (!Boolean.TRUE.equals(hasQualifyingSubmit.value())) {
                return;
            }

            Long max = maxStageCount.value();
            if (max == null) {
                return;
            }

            out.collect(Tuple2.of(ctx.getCurrentKey(), max));
        }
    }

}
