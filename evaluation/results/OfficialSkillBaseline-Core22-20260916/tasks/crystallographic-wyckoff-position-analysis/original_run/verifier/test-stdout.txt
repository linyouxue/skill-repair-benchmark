from __future__ import annotations

from collections import Counter
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _as_float_in_unit_interval(x: float) -> float:
    x = float(x) % 1.0
    if abs(x - 1.0) < 1e-12:
        return 0.0
    if abs(x) < 1e-12:
        return 0.0
    return x


def _float_to_rational_str(x: float, *, max_denominator: int = 12) -> str:
    x = _as_float_in_unit_interval(x)
    frac = Fraction(x).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def _get_wyckoffs_from_dataset(dataset: Any) -> list[str]:
    if dataset is None:
        raise ValueError("No symmetry dataset returned by spglib")

    if isinstance(dataset, dict):
        wyckoffs = dataset.get("wyckoffs")
    else:
        wyckoffs = getattr(dataset, "wyckoffs", None)

    if wyckoffs is None:
        raise ValueError("Symmetry dataset does not contain Wyckoff labels")

    return [str(w) for w in wyckoffs]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff multiplicities and first-site coordinates from a CIF.

    Returns a dict with keys:
      - wyckoff_multiplicity_dict: {wyckoff_letter: multiplicity_in_cell}
      - wyckoff_coordinates_dict: {wyckoff_letter: [x, y, z]} as rationals

    If analysis fails, returns a dict containing an "error" key.
    """

    try:
        structure = Structure.from_file(filepath)
        if not structure.is_ordered:
            structure = structure.get_ordered_structure()

        dataset = None
        for symprec in (1e-3, 1e-2, 5e-2, 1e-1):
            try:
                sga = SpacegroupAnalyzer(
                    structure, symprec=symprec, angle_tolerance=5.0
                )
                dataset = sga.get_symmetry_dataset()
                wyckoffs = _get_wyckoffs_from_dataset(dataset)
                if len(wyckoffs) != len(structure):
                    raise ValueError(
                        "Wyckoff label list length does not match number of sites"
                    )
                break
            except Exception:
                dataset = None

        if dataset is None:
            raise ValueError("Failed to obtain symmetry dataset from structure")

        wyckoffs = _get_wyckoffs_from_dataset(dataset)

        multiplicities = Counter(wyckoffs)
        multiplicity_dict: dict[str, int] = dict(multiplicities)

        coordinates_dict: dict[str, list[str]] = {}
        for i, letter in enumerate(wyckoffs):
            if letter in coordinates_dict:
                continue
            coords = structure[i].frac_coords
            coordinates_dict[letter] = [
                _float_to_rational_str(float(c), max_denominator=12) for c in coords
            ]

        return {
            "wyckoff_multiplicity_dict": multiplicity_dict,
            "wyckoff_coordinates_dict": coordinates_dict,
        }
    except Exception as exc:
        return {"error": str(exc), "filepath": filepath}
